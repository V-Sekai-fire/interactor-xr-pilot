# The QUIC edge: picoquic and picotls (third_party subtrees) over OpenSSL libcrypto.
# Configure with -DOPENSSL_ROOT_DIR. smoke is the link gate; a bad certificate path is its control.
cmake_minimum_required(VERSION 3.20)
project(xr_pilot_edge C)
set(PQ ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/picoquic)
set(PT ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/picotls)
file(GLOB CORE ${PQ}/picoquic/*.c)
list(FILTER CORE EXCLUDE REGEX ".*(picoquic_ptls_fusion|picoquic_mbedtls)\\.c$")
file(GLOB HTTP ${PQ}/picohttp/*.c)
file(GLOB LOGLIB ${PQ}/loglib/*.c)
set(TLS ${PT}/lib/picotls.c ${PT}/lib/pembase64.c ${PT}/lib/hpke.c ${PT}/lib/asn1.c ${PT}/lib/openssl.c)
file(GLOB CIFRA_SRC ${PT}/deps/cifra/src/aes.c ${PT}/deps/cifra/src/blockwise.c ${PT}/deps/cifra/src/chacha20.c ${PT}/deps/cifra/src/chash.c ${PT}/deps/cifra/src/curve25519.c ${PT}/deps/cifra/src/drbg.c ${PT}/deps/cifra/src/hmac.c ${PT}/deps/cifra/src/gcm.c ${PT}/deps/cifra/src/gf128.c ${PT}/deps/cifra/src/modes.c ${PT}/deps/cifra/src/poly1305.c ${PT}/deps/cifra/src/sha256.c ${PT}/deps/cifra/src/sha512.c)
set(MINI ${PT}/deps/micro-ecc/uECC.c ${PT}/lib/cifra.c ${PT}/lib/cifra/x25519.c ${PT}/lib/cifra/chacha20.c ${PT}/lib/cifra/aes128.c ${PT}/lib/cifra/aes256.c ${PT}/lib/cifra/random.c ${PT}/lib/minicrypto-pem.c ${PT}/lib/uecc.c ${PT}/lib/ffx.c)
if(NOT WIN32)
  list(FILTER CORE EXCLUDE REGEX ".*winsockloop\\.c$")
endif()
add_library(pq STATIC ${CORE} ${HTTP} ${LOGLIB} ${TLS} ${CIFRA_SRC} ${MINI})
target_include_directories(pq PUBLIC ${PT}/include ${PQ}/picoquic ${PQ}/picohttp ${PQ}/loglib ${OPENSSL_ROOT_DIR}/include PRIVATE ${PT}/deps/cifra/src/ext ${PT}/deps/cifra/src ${PT}/deps/micro-ecc)
target_compile_definitions(pq PUBLIC PTLS_WITHOUT_FUSION DISABLE_DEBUG_PRINTF)
if(WIN32)
  target_include_directories(pq BEFORE PUBLIC ${PT}/picotlsvs/picotls)
  target_compile_definitions(pq PUBLIC _WINDOWS)
  target_compile_options(pq PRIVATE -include errno.h -w -Wno-error=incompatible-pointer-types)
  target_link_libraries(pq PUBLIC ws2_32 iphlpapi bcrypt)
endif()
add_executable(smoke smoke.c)
find_library(CRYPTO_LIB NAMES crypto libcrypto PATHS ${OPENSSL_ROOT_DIR}/lib ${OPENSSL_ROOT_DIR}/lib64 NO_DEFAULT_PATH)
target_link_libraries(pq PUBLIC ${CRYPTO_LIB})
target_link_libraries(smoke pq)
if(WIN32)
  target_sources(smoke PRIVATE edge_win.c)
  target_link_libraries(smoke crypt32 ws2_32)
  target_link_options(smoke PRIVATE -static -Wl,--allow-multiple-definition)
endif()
find_package(Threads REQUIRED)
add_executable(mcp_edge mcp_edge.c)
target_link_libraries(mcp_edge pq Threads::Threads)
if(WIN32)
  target_sources(mcp_edge PRIVATE edge_win.c)
  target_link_libraries(mcp_edge crypt32 ws2_32)
  target_link_options(mcp_edge PRIVATE -static -Wl,--allow-multiple-definition)
endif()
